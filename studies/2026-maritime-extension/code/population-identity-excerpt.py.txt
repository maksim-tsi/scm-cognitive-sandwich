def _replace_identity(value: Any, old: str, new: str) -> Any:
    if isinstance(value, dict):
        return {key: _replace_identity(item, old, new) for key, item in value.items()}
    if isinstance(value, list):
        return [_replace_identity(item, old, new) for item in value]
    if isinstance(value, str):
        return value.replace(old, new)
    return value


def scientific_population() -> list[dict[str, Any]]:
    units: list[dict[str, Any]] = []
    for index, source in enumerate(ORIGINAL_POPULATION(), 1):
        old_incident = str(source["incident_id"])
        incident_id = f"{IDENTITY_PREFIX}{index:03d}"
        item = deepcopy(source)
        item = _replace_identity(item, old_incident, incident_id)
        if not isinstance(item, dict):
            raise RuntimeError("scientific identity replacement failed")
        item["incident_id"] = incident_id
        item["unit_id"] = f"{incident_id}-GEN-001"
        item["generation_id"] = 1
        item["input_payload_sha256"] = sha_bytes(canonical(item["input"]).encode())
        units.append(item)
    strata: dict[str, int] = {}
    for unit in units:
        stratum = str(unit["stratum_id"])
        strata[stratum] = strata.get(stratum, 0) + 1
    if len(units) != 60 or len({unit["unit_id"] for unit in units}) != 60:
        raise RuntimeError("scientific population denominator or identity drift")
    if len(strata) != 4 or set(strata.values()) != {15}:
        raise RuntimeError("scientific population is not four-stratum balanced")
    return units
